cmake_minimum_required(VERSION 3.14)
project(abak_cps_signature_tests LANGUAGES CXX)
enable_testing()
set(CPS "${CMAKE_CURRENT_LIST_DIR}/../runner/cps")
add_executable(cps_signature_tests signature_test.cpp "${CPS}/cps_profile_reader.cpp"
    "${CPS}/cps_signature.cpp" "${CPS}/cps_wire.cpp")
set(SIGNATURE_TARGETS cps_signature_tests)
add_test(NAME cps_signature COMMAND cps_signature_tests)
if(WIN32)
  add_executable(cps_public_crypto_tests public_crypto_test.cpp "${CPS}/cps_crypto_windows.cpp"
      "${CPS}/cps_signature.cpp" "${CPS}/cps_profile_reader.cpp")
  target_link_libraries(cps_public_crypto_tests PRIVATE Crypt32 Bcrypt)
  # Fake worker is built in this test folder only and is never installed.
  add_executable(ABAKCpsReader "${CPS}/cps_worker_main.cpp" "${CPS}/cps_wire.cpp"
      "${CPS}/cps_profile_reader.cpp" "${CPS}/cps_signature.cpp" fake_signing_reader.cpp)
  target_link_libraries(ABAKCpsReader PRIVATE Wer)
  add_executable(cps_signature_process_tests process_signature_test.cpp
      "${CPS}/cps_windows_client.cpp" "${CPS}/cps_profile_reader.cpp" "${CPS}/cps_wire.cpp")
  target_link_libraries(cps_signature_process_tests PRIVATE Shell32 Ole32 Uuid)
  add_dependencies(cps_signature_process_tests ABAKCpsReader)
  add_test(NAME cps_signature_process COMMAND cps_signature_process_tests)
  set_tests_properties(cps_signature_process PROPERTIES TIMEOUT 30)
  list(APPEND SIGNATURE_TARGETS cps_public_crypto_tests ABAKCpsReader cps_signature_process_tests)
  add_test(NAME cps_public_crypto COMMAND cps_public_crypto_tests)
endif()
foreach(target ${SIGNATURE_TARGETS})
  target_compile_features(${target} PRIVATE cxx_std_17)
  target_include_directories(${target} PRIVATE "${CPS}")
  if(WIN32)
    target_compile_definitions(${target} PRIVATE NOMINMAX WIN32_LEAN_AND_MEAN)
  endif()
  if(MSVC)
    target_compile_options(${target} PRIVATE /W4 /WX /EHsc)
  else()
    target_compile_options(${target} PRIVATE -Wall -Wextra -Werror)
  endif()
endforeach()
