cmake_minimum_required(VERSION 3.14)
project(abak_cda_reports_tests LANGUAGES CXX)
enable_testing()
add_executable(cda_reports_tests test.cpp ../document_controls_runtime/cda_reports.cpp)
target_compile_features(cda_reports_tests PRIVATE cxx_std_17)
if(MSVC)
  target_compile_options(cda_reports_tests PRIVATE /W4 /WX /EHsc /utf-8)
else()
  target_compile_options(cda_reports_tests PRIVATE -Wall -Wextra -Werror -pedantic)
endif()
add_test(NAME cda_reports COMMAND cda_reports_tests)
