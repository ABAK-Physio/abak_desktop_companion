cmake_minimum_required(VERSION 3.14)
project(abak_cps_tests LANGUAGES CXX)
enable_testing()
set(CPS "${CMAKE_CURRENT_LIST_DIR}/../runner/cps")
add_executable(cps_profile_tests profile_test.cpp "${CPS}/cps_profile_reader.cpp")
add_executable(cps_wire_tests wire_test.cpp "${CPS}/cps_profile_reader.cpp" "${CPS}/cps_wire.cpp")
set(CPS_TEST_TARGETS cps_profile_tests cps_wire_tests)
add_test(NAME cps_profile COMMAND cps_profile_tests)
add_test(NAME cps_wire COMMAND cps_wire_tests)
if(WIN32)
  # Only this test executable uses a fake provider; it is never installed.
  add_executable(ABAKCpsReader "${CPS}/cps_worker_main.cpp" "${CPS}/cps_wire.cpp"
      "${CPS}/cps_profile_reader.cpp" fake_reader.cpp)
  add_executable(cps_process_tests process_test.cpp "${CPS}/cps_windows_client.cpp"
      "${CPS}/cps_profile_reader.cpp" "${CPS}/cps_wire.cpp")
  add_dependencies(cps_process_tests ABAKCpsReader)
  target_link_libraries(ABAKCpsReader PRIVATE Wer)
  target_link_libraries(cps_process_tests PRIVATE Shell32 Ole32 Uuid)
  list(APPEND CPS_TEST_TARGETS ABAKCpsReader cps_process_tests)
  add_test(NAME cps_process COMMAND cps_process_tests)
  set_tests_properties(cps_process PROPERTIES TIMEOUT 20)
endif()
foreach(target ${CPS_TEST_TARGETS})
  target_compile_features(${target} PRIVATE cxx_std_17)
  target_include_directories(${target} PRIVATE "${CPS}")
  if(WIN32)
    target_compile_definitions(${target} PRIVATE NOMINMAX WIN32_LEAN_AND_MEAN)
  endif()
  if(MSVC)
    target_compile_options(${target} PRIVATE /W4 /WX /EHsc)
  endif()
endforeach()
